\olsection{Simple Quantified Modal Logic Semantics}

The Simple Quantified Modal Logic is called \Log{SQML}.

\begin{defn}[\Log{SQML} Models]
    A \Log{SQML} \emph{model} $\Struct {M}$ is a triple
    $\tuple{W, D, I}$, where
    \begin{itemize}
    \item $W$ is a nonempty set (``worlds''),
    \item $D$ is a nonempty set (``individuals''),
    \item $I$ is a function assigning
    \begin{itemize}
        \item to each !!{constant}~$c$ of $\Lang{L}$, !!a{element}
        $\Assign{c}{M}\in D$,
        \item to each $n$-place !!{predicate}~$R$ of $\Lang{L}$ 
        (other than $\eq$), an $n+1$-place relation $\Assign{R}{M}$ over 
        $D^n\times W$,
        % \item to each $n$-place
        % !!{function}~$f$ of $\Lang{L}$, an $n+1$-place function $\Assign{f}{M}
        % \colon D^n\times W  \to D$.    
    \end{itemize}
    \end{itemize}
    A \emph{pointed} \Log{SQML} model is a quadruple $\tuple{W, @, D, I}$
    with $@\in W$ (the model's `actual world').
\end{defn}

\Log{SQML} models don't have an accessibility relation. The semantics
will in effect treat each world as accessible from any world. 

Variable assignments and the value of terms are defined as in first
order logic:

\begin{defn}[Variable Assignment]
    A \emph{variable assignment}~$s$ for a $\Log{SQML}$ model with
    domain $D$ is a function that assigns to each !!{variable}
    !!a{element} of~$D$, i.e., $s\colon \Var \to D$.
\end{defn}


\begin{defn}[$x$-Variant]
    If $s$ is !!a{variable} assignment for a $\Log{SQML}$ model $\Struct M$ with 
    domain $D$, then any
    !!{variable} assignment~$s'$ for~$\Struct M$ which differs from~$s$ at most
    in what it assigns to~$x$ is called an \emph{$x$-variant} of~$s$.  If
    $s'$ is an $x$-variant of~$s$ we write $\varAssign{s'}{s}{x}$.
\end{defn}


\begin{defn}[!!^{value} of Terms]
    If $t$ is a term of the language~$\Lang L$, $\Struct M$ is a 
    \Log{SQML} model, and $s$ is !!a{variable} assignment
    for~$\Struct M$, the \emph{!!{value}}~$\Value{t}{M}[s]$ is defined as
    follows:
    \begin{enumerate}
    \item \indcase{t}{c}{$\Value{\indfrm}{M}[s] = \Assign{\indcomplex}{M}$.}
    \item \indcase{t}{x}{$\Value{\indfrm}{M}[s] = s(\indcomplex)$.}
    % \item \indcase{t}{\Atom{f}{t_1, \ldots, t_n}}{
    % \[
    % \Value{\indfrm}{M}[s] = \Assign{f}{M}(\Value{t_1}{M}[s], \ldots,
    % \Value{t_n}{M}[s]).
    % \]}
    \end{enumerate}
\end{defn}

The semantics combines clauses from modal logic and first order logic.

\begin{defn}[\Log{SQML} Satisfaction]\ollabel{defn:sqml-satis}
    Satisfaction of a !!{formula}~$!A$ in a \Log{SQML} model $\Struct
    M$ relative to a world $w$ and a !!{variable} assignment~$s$, in
    symbols: $\Sat{M,w}{!A}[s]$, is defined thus:
    \begin{enumerate}
    \tagitem{prvFalse}{%
      \indcase{!A}{\lfalse}{$\Sat/{M,w}{\indfrm}[s]$.}}{}
    
    \tagitem{prvTrue}{%
      \indcase{!A}{\ltrue}{$\Sat{M,w}{\indfrm}[s]$.}}{}
    
    \item \indcase{!A}{\Atom{R}{t_1, \dots, t_n}}{$\Sat{M,w}{\indfrm}[s]$
      iff $\langle \Value{t_1}{M}[s], \dots, \Value{t_n}{M}[s], w \rangle \in
      \Assign{R}{M}$.}
    
    \item \indcase{!A}{\eq[t_1][t_2]}{$\Sat{M,w}{\indfrm}[s]$ iff
      $\Value{t_1}{M}[s] = \Value{t_2}{M}[s]$.}
    
    \tagitem{prvNot}{%
      \indcase{!A}{\lnot !B}{$\Sat{M,w}{\indfrm}[s]$ iff
        $\Sat/{M,w}{!B}[s]$.}}{}
    
    \tagitem{prvAnd}{%
      \indcase{!A}{(!B \land !C)}{$\Sat{M,w}{\indfrm}[s]$ iff $\Sat{M,w}{!B}[s]$
        and $\Sat{M,w}{!C}[s]$.}}{}
    
    \tagitem{prvOr}{%
      \indcase{!A}{(!B \lor !C)}{$\Sat{M,w}{\indfrm}[s]$ iff
        $\Sat{M,w}{!A}[s]$ or $\Sat{M,w}{!B}[s]$ (or both).}}{}
    
    \tagitem{prvIf}{%
      \indcase{!A}{(!B \lif !C)}{$\Sat{M,w}{\indfrm}[s]$ iff $\Sat/{M,w}{!B}[s]$
        or $\Sat{M,w}{!C}[s]$ (or both).}}{}
    
    \tagitem{prvIff}{%
      \indcase{!A}{(!B \liff !C)}{$\Sat{M,w}{\indfrm}[s]$ iff either both
        $\Sat{M,w}{!B}[s]$ and $\Sat{M,w}{!C}[s]$, or neither.}}{}
    
    \tagitem{prvAll}{%
      \indcase{!A}{\lforall[x][!B]}{$\Sat{M,w}{\indfrm}[s]$ iff for every
        $x$-variant~$s'$ of $s$, $\Sat{M,w}{!B}[s']$.}}{}
    
    \tagitem{prvEx}{%
      \indcase{!A}{\lexists[x][!B]}{$\Sat{M,w}{\indfrm}[s]$ iff there is an
        $x$-variant~$s'$ of $s$ so that $\Sat{M,w}{!B}[s']$.}}{}
    
    \item\indcase{!A}{\Box !B}{$\Sat{M,w}{\indfrm}[s]$ iff
        $\Sat{M,w'}{!B}[s]$ for all $w' \in W$.}
    \item\indcase{!A}{\Diamond !B}{$\Sat{M,w}{\indfrm}[s]$ iff
        $\Sat{M,w'}{!B}[s]$ for at least one $w' \in W$.}
      \end{enumerate}
\end{defn}

Other semantic notions are defined as you would expect.

If $!A$ is a sentence, we say that $!A$ is \emph{true at $w$} in 
$\Struct M$ iff it is true at $w$ relative to some assignment, or,
equivalent, true at $w$ relative to any assignment.

If $\Struct M$ is a pointed model and $!A$ a sentence, we say that
$!A$ is \emph{true in} $\Struct M$ if it's true in the model's actual
world $@$.

A sentence is a \emph{logical truth} of $\Log{SQML}$ iff it is true at
every world of every $\Log{SQML}$~model (or, equivalently true at
every pointed $\Log{SQML}$~model). An argument is \emph{valid} in
$\Log{SQML}$ iff there is no world in any $\Log{SQML}$~model relative to
which all of its premises are true but its conclusion false.
